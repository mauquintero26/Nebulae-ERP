4. TABLAS Y RELACIONES
===================================================================

  customer_request_lines
    cr_id -> customer_requests.id, sku_id -> product_skus.id

  sales_quotation_lines
    sq_id, cr_line_id (nullable), sku_id

  sale_order_lines_erp
    so_id, sq_line_id (nullable), sku_id

  purchase_order_lines
    pec_id -> purchase_orders_full.id, sku_id

  procurement_allocations
    po_line_id, sale_order_line_id (nullable)
    allocation_type: CUSTOMER_ORDER | NEBULAE_STOCK | MAU_STOCK
    REGLA: CUSTOMER_ORDER exige sale_order_line_id
    UNIQUE(po_line_id, sale_order_line_id) WHERE NOT NULL

  goods_receipt_lines
    gr_id, po_line_id (nullable), sku_id
    quantity_expected, quantity_received, quantity_rejected, quantity_quarantine
    source: NATIVE | BACKFILL

  goods_receipt_line_allocations
    gr_line_id, allocation_id — UNIQUE(gr_line_id, allocation_id)

  inventory_owner_balances
    sku_id, warehouse_id, owner (NEBULAE|MAU) — UNIQUE(sku, wh, owner)

  inventory_reservations
    sku_id, warehouse_id, sale_order_line_id, status, expires_at
    status: ACTIVE | RELEASED | CONVERTED | EXPIRED

  payment_transactions
    tipo: ANTICIPO|ABONO|SALDO|DEVOLUCION|REVERSION
    currency: COP|USD, monto, referencia, metodo_pago
    reversed_by_id (FK self), is_reversed, user_name, created_at
